/* Ruimt de service worker van de oude app op, zodat telefoons de verhuispagina
   zien in plaats van de oude app uit hun cache. Raakt de opgeslagen gegevens
   (IndexedDB scc_db, localStorage) niet aan. */
self.addEventListener('install',()=>self.skipWaiting());
self.addEventListener('activate',e=>e.waitUntil((async()=>{
  const keys=await caches.keys();
  await Promise.all(keys.map(k=>caches.delete(k)));
  await self.registration.unregister();
  const cs=await self.clients.matchAll({type:'window'});
  cs.forEach(c=>{try{c.navigate(c.url);}catch(e){}});
})()));
